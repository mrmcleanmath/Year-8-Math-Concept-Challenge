function shuffle(items){
  const a = [...items];
  for(let i=a.length-1;i>0;i--){
    const j = Math.floor(Math.random()*(i+1));
    [a[i],a[j]]=[a[j],a[i]];
  }
  return a;
}
function escapeHtml(value){
  return String(value)
    .replaceAll("&","&amp;")
    .replaceAll("<","&lt;")
    .replaceAll(">","&gt;")
    .replaceAll('"',"&quot;");
}
function setFeedback(kind,en,sv){
  feedback.className = `feedback ${kind} show`;
  feedbackIcon.textContent = kind === "good" ? "✓" : "×";
  feedbackCopy.innerHTML = `${escapeHtml(en)}<span class="feedback-sv" lang="sv">${escapeHtml(sv)}</span>`;
}
function clearFeedback(){
  feedback.className = "feedback";
  feedbackIcon.textContent = "";
  feedbackCopy.textContent = "";
}
function updateHud(){
  const total = round.length || QUIZ_SIZE;
  const completed = Math.min(index,total);
  const pct = total ? Math.round((completed/total)*100) : 0;
  progressText.textContent = index < total ? `Question ${index+1} of ${total}` : `${total} of ${total} completed`;
  progressPercent.textContent = `${pct}%`;
  progressFill.style.width = `${pct}%`;
  completedStat.textContent = completed;
  streakStat.textContent = streak;
  wrongStat.textContent = wrongAttempts;
}
function burst(){
  celebrationLayer.innerHTML = "";
  const vectors = [
    [-155,-80,-150],[-120,-118,-100],[-72,-138,-55],[-18,-150,-20],[42,-142,35],[98,-118,85],[148,-72,125],
    [-164,-16,-85],[-142,42,-55],[-92,88,-20],[-35,112,30],[28,116,75],[88,92,115],[145,45,150]
  ];
  vectors.forEach((v,i)=>{
    const s=document.createElement("span");
    s.className="spark";
    s.style.setProperty("--x",`${v[0]}px`);
    s.style.setProperty("--y",`${v[1]}px`);
    s.style.setProperty("--r",`${v[2]}deg`);
    s.style.width=`${6+(i%3)*2}px`;
    s.style.height=`${6+((i+1)%3)*2}px`;
    s.style.borderRadius=i%2 ? "50%" : "2px";
    if(i%3===1) s.style.background="#5f86e4";
    if(i%3===2) s.style.background="#8ab6ff";
    celebrationLayer.append(s);
  });
  setTimeout(()=>{celebrationLayer.innerHTML=""},850);
}
function renderQuestion(){
  if(index >= round.length){finishRound();return}
  locked=false;
  missedCurrent=false;
  clearFeedback();
  const item=round[index];
  questionBadge.textContent=`${index+1} / ${round.length}`;
  questionEn.textContent=item.en;
  questionSv.textContent=item.sv;
  optionsEl.innerHTML="";
  const choices=shuffle([
    {...item.correct,correct:true},
    ...item.wrong.map(answer=>({...answer,correct:false}))
  ]);
  const letters=["A","B","C","D"];
  choices.forEach((choice,i)=>{
    const button=document.createElement("button");
    button.type="button";
    button.className="option";
    button.dataset.correct=choice.correct ? "true" : "false";
    button.innerHTML=`
      <span class="option-letter">${letters[i]}</span>
      <span class="option-copy">
        <span class="option-en">${escapeHtml(choice.en)}</span>
        <span class="option-sv" lang="sv">${escapeHtml(choice.sv)}</span>
      </span>`;
    button.addEventListener("click",()=>chooseAnswer(button));
    optionsEl.append(button);
  });
  updateHud();
}
function chooseAnswer(button){
  if(locked || button.disabled) return;
  if(button.dataset.correct === "true"){
    locked=true;
    button.classList.add("correct");
    optionsEl.querySelectorAll(".option").forEach(btn=>btn.disabled=true);
    if(missedCurrent){
      streak=0;
    }else{
      streak++;
      bestStreak=Math.max(bestStreak,streak);
    }
    setFeedback("good","Correct!","Rätt!");
    burst();
    updateHud();
    advanceTimer=setTimeout(()=>{
      advanceTimer=null;
      index++;
      renderQuestion();
    },1050);
  }else{
    wrongAttempts++;
    missedCurrent=true;
    streak=0;
    button.classList.add("wrong");
    button.disabled=true;
    setFeedback("bad","Try again.","Försök igen.");
    updateHud();
  }
}
function finishRound(){
  questionView.style.display="none";
  completion.classList.add("show");
  resultCompleted.textContent=round.length;
  resultWrong.textContent=wrongAttempts;
  resultStreak.textContent=bestStreak;
  if(wrongAttempts===0){
    completionMessage.textContent=`Perfect round — all ${round.length} questions completed with no wrong attempts.`;
    burst();
  }else{
    completionMessage.textContent=`You completed all ${round.length} questions with ${wrongAttempts} wrong ${wrongAttempts===1 ? "attempt" : "attempts"}.`;
  }
  index=round.length;
  updateHud();
}
function newRound(){
  if(advanceTimer){clearTimeout(advanceTimer);advanceTimer=null}
  const bank = QUESTION_BANKS[selectedChapter] || QUESTION_BANKS["1.1"];
  const section = SECTIONS[selectedChapter] || SECTIONS["1.1"];
  sectionName.textContent = section.en;
  sectionSv.textContent = section.sv;
  chapterButtons.forEach(btn=>btn.classList.toggle("active",btn.dataset.chapter===selectedChapter));
  const questionCount = Math.min(QUIZ_SIZE,bank.length);
  roundChip.textContent = bank.length > questionCount ? `${questionCount} random questions` : `${questionCount} questions`;
  round=shuffle(bank).slice(0,questionCount);
  index=0;
  wrongAttempts=0;
  streak=0;
  bestStreak=0;
  locked=false;
  missedCurrent=false;
  completion.classList.remove("show");
  questionView.style.display="block";
  renderQuestion();
}

  chapterMenu.classList.toggle("show",open);
  chapterTrigger.setAttribute("aria-expanded",open ? "true" : "false");

newRoundBtn.addEventListener("click",newRound);
playAgainBtn.addEventListener("click",newRound);
root.addEventListener("keydown",e=>{
  if(e.target.matches("button")) return;
});
newRound();
